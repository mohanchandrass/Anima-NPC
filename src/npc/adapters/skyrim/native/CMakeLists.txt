cmake_minimum_required(VERSION 3.20)
set(CMAKE_NINJA_CMCLDEPS_RC OFF)
project(AnimaConnector VERSION 0.5.3 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AnimaConnector requires Windows x64")
endif()
set(ANIMA_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../../../.tools" CACHE PATH "Pinned local checkouts")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
# The connector uses event sinks, not instruction-patching trampolines.
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)
set(COMMONLIB_PREBUILT OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${ANIMA_DEPS}/fmt" "${CMAKE_BINARY_DIR}/fmt")
set(fmt_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake")
add_subdirectory("${ANIMA_DEPS}/spdlog" "${CMAKE_BINARY_DIR}/spdlog")
set(spdlog_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake")
set(RAPIDCSV_INCLUDE_DIRS "${ANIMA_DEPS}/rapidcsv/src" CACHE PATH "" FORCE)
add_subdirectory("${ANIMA_DEPS}/DirectXMath" "${CMAKE_BINARY_DIR}/directxmath")
add_library(Microsoft::DirectXMath ALIAS DirectXMath)
set(directxmath_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake")
# CommonLib uses only SimpleMath types in BSGraphics::State. This plugin does
# not call DirectXTK rendering/audio functions or need its HLSL shader compiler.
add_library(Microsoft::DirectXTK INTERFACE IMPORTED)
set_target_properties(Microsoft::DirectXTK PROPERTIES
  INTERFACE_INCLUDE_DIRECTORIES "${ANIMA_DEPS}/DirectXTK/Inc")
set(directxtk_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake")
add_subdirectory("${ANIMA_DEPS}/CommonLibSSE-NG" "${CMAKE_BINARY_DIR}/commonlib")
target_link_libraries(CommonLibSSE PUBLIC Microsoft::DirectXMath)
add_library(AnimaConnector SHARED plugin.cpp)
target_include_directories(AnimaConnector PRIVATE "${ANIMA_DEPS}/json/single_include" "${RAPIDCSV_INCLUDE_DIRS}")
target_link_libraries(AnimaConnector PRIVATE CommonLibSSE::CommonLibSSE winhttp)
target_compile_definitions(AnimaConnector PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AnimaConnector PRIVATE /utf-8 /EHsc /W4)
install(TARGETS AnimaConnector RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
install(FILES AnimaConnector.ini DESTINATION SKSE/Plugins)
enable_language(RC)
add_executable(AnimaConsole console/AnimaConsole.cpp console/AnimaConsole.rc)
target_compile_definitions(AnimaConsole PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(AnimaConsole PRIVATE /utf-8 /EHsc /W4)
target_link_options(AnimaConsole PRIVATE /SUBSYSTEM:CONSOLE)
enable_testing()
add_executable(AnimaDeliveryTests "${CMAKE_CURRENT_LIST_DIR}/../../../../../tests/unit/adapters/skyrim_delivery.cpp")
target_include_directories(AnimaDeliveryTests PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
add_test(NAME skyrim_delivery COMMAND AnimaDeliveryTests)

add_executable(AnimaRuntimeTests "${CMAKE_CURRENT_LIST_DIR}/../../../../../tests/unit/adapters/skyrim_runtime.cpp")
target_include_directories(AnimaRuntimeTests PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
add_test(NAME skyrim_runtime COMMAND AnimaRuntimeTests)

target_link_libraries(AnimaRuntimeTests PRIVATE CommonLibSSE::CommonLibSSE)
